cmake_minimum_required(VERSION 3.16)
project(HPCParallelization LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
add_library(poisson_core src/poisson.cpp)
target_include_directories(poisson_core PUBLIC include)
target_compile_options(poisson_core PRIVATE -Wall -Wextra -Wpedantic $<$<CONFIG:Release>:-O3>)
add_executable(poisson_serial src/serial_main.cpp)
target_link_libraries(poisson_serial PRIVATE poisson_core)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME serial_correctness COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_serial.py $<TARGET_FILE:poisson_serial>)
endif()
