cmake_minimum_required(VERSION 3.16)
project(HPCParallelization LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
add_library(poisson_core src/poisson.cpp)
target_include_directories(poisson_core PUBLIC include)
target_compile_options(poisson_core PRIVATE -Wall -Wextra -Wpedantic $<$<CONFIG:Release>:-O3>)
add_executable(poisson_serial src/serial_main.cpp)
target_link_libraries(poisson_serial PRIVATE poisson_core)
option(BUILD_MPI "Build the MPI Poisson solver" OFF)
if(BUILD_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_library(poisson_mpi_core src/mpi_poisson.cpp)
  target_link_libraries(poisson_mpi_core PUBLIC poisson_core MPI::MPI_CXX)
  target_compile_options(poisson_mpi_core PRIVATE -Wall -Wextra -Wpedantic $<$<CONFIG:Release>:-O3>)
  add_executable(poisson_mpi src/mpi_main.cpp)
  target_link_libraries(poisson_mpi PRIVATE poisson_mpi_core)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME serial_correctness COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_serial.py $<TARGET_FILE:poisson_serial>)
  foreach(suite mpi_workflow mpi_launcher pbs_allocation analyze_mpi)
    add_test(NAME ${suite}_tooling COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${suite}.py)
  endforeach()
endif()
